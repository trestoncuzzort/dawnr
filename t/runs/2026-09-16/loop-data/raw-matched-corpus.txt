    while rem >= m
      invariant n == q * m + rem
      invariant rem >= 0
      decreases rem
    {
      q := q + 1;
      rem := rem - m;
    }
  } else {
    while rem < 0
      invariant n == q * m + rem
      invariant rem < m
      decreases -rem
    {
